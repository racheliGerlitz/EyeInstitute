-- בסיס ללולאה חוזרת לרופא מסוים
DECLARE @doctorId INT
DECLARE @allowedDays TABLE (DoctorId INT, Weekday INT)

-- הכנסת ימי עבודה לרופאים
INSERT INTO @allowedDays VALUES
(11111111, 1), (11111111, 2), (11111111, 5),
(22222222, 1), (22222222, 2), (22222222, 4),
(33333333, 1), (33333333, 5), (33333333, 6),
(44444444, 1), (44444444, 2), (44444444, 3), (44444444, 5),
(55555555, 2), (55555555, 3), (55555555, 4),
(66666666, 1), (66666666, 5),
(77777777, 1), (77777777, 6),
(88888888, 1), (88888888, 2), (88888888, 5),
(99999999, 1), (99999999, 4), (99999999, 5)

-- טבלת הרופאים
DECLARE @doctors TABLE (DoctorId INT)
INSERT INTO @doctors VALUES
(11111111), (22222222), (33333333), (44444444),
(55555555), (66666666), (77777777), (88888888), (99999999)

-- לולאה עבור כל רופא
DECLARE doctor_cursor CURSOR FOR SELECT DoctorId FROM @doctors
OPEN doctor_cursor
FETCH NEXT FROM doctor_cursor INTO @doctorId

WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @counter INT = 0
    DECLARE @date DATE = '2025-06-01'

    WHILE @counter < 100
    BEGIN
        -- אם יום העבודה תואם לרופא
        IF EXISTS (
            SELECT 1 FROM @allowedDays
            WHERE DoctorId = @doctorId AND Weekday = DATEPART(WEEKDAY, @date)
        )
        BEGIN
            DECLARE @timeSlot INT = 0
            WHILE @timeSlot < 16 AND @counter < 100
            BEGIN
                DECLARE @dateTime DATETIME = DATEADD(MINUTE, @timeSlot * 15, DATEADD(HOUR, 9, CAST(@date AS DATETIME)))

                INSERT INTO Appointments (Date, Hour, DoctorId, ClientId)
                VALUES (@dateTime, DATEPART(HOUR, @dateTime), @doctorId, NULL)

                SET @counter += 1
                SET @timeSlot += 1
            END
        END

        SET @date = DATEADD(DAY, 1, @date)
    END

    FETCH NEXT FROM doctor_cursor INTO @doctorId
END

CLOSE doctor_cursor
DEALLOCATE doctor_cursor
select* from Appointments
